## 熵

$$
\begin{aligned}
H(X) &= -\sum p(a_i) \log_2 p(a_i) \\
H(XY) &= -\sum\sum p(a_i b_j) \log_2 p(a_i b_j) = H(X) + H(Y \mid X) = H(Y) + H(X \mid Y) \\
H(Y \mid X) &= -\sum\sum p(a_i b_j) \log_2 p(b_j \mid a_i) = H(XY) - H(X) \\
I(X;Y) &= \sum\sum p(a_i b_j) \log_2\frac{p(a_i b_j)}{p(a_i)p(b_j)} = H(X) - H(X \mid Y) = H(X) + H(Y) - H(XY) \\
H(X^2) &= 2H(X),\; H(X^3) = 3H(X) \quad \text{(无记忆扩展)} \\
H(X \mid Y) &\le H(X),\; H(X) \le \log_2 n \quad \text{(极值性; 最大熵定理)} \\
H(XY) &= H(X) + H(Y) \quad (X, Y\text{独立时}) \\
\text{熵函数上凸}&:\; H(\alpha P + (1-\alpha)Q) \ge \alpha H(P) + (1-\alpha) H(Q) \\
\text{连续(限峰值)}&:\; H_c(X) = \log_2(b-a) \quad \text{(均匀分布最大)} \\
\text{连续(限功率)}&:\; H_c(X) = \frac{1}{2}\log_2(2\pi e\sigma^2) \quad \text{(高斯分布最大)}
\end{aligned}
$$

## 信道容量

$$
\begin{aligned}
C &= \max_{p(a_i)} I(X;Y) \\
\text{一一对应无噪}&:\; C = \log_2 n \\
\text{扩展无噪}&:\; C = \log_2 n \\
\text{归并无噪}&:\; C = \log_2 m \\
\text{强对称}(n=m)&:\; C = \log_2 n - H(p) - p \cdot \log_2(n-1) \\
\text{对称信道}&:\; C = \log_2 m - H(\text{行元素}) \\
\text{二进制均匀}&:\; C = 1 - H(p) \\
\text{连续高斯}&:\; C = W \cdot \log_2(1 + S/N) \quad [S/N\text{为功率比}] \\
S/N(\mathrm{dB}) &= 10 \cdot \log_{10}(S/N),\; S/N = 10^{\mathrm{dB}/10} \\
\text{有噪编码定理}&:\; R < C \to L\text{足够大时 }P_e \to 0;\; R > C \to P_e > 0
\end{aligned}
$$

## 信源编码

$$
\begin{aligned}
\bar{K} &= \sum p(a_i) \cdot k_i \\
\eta &= H(X) / \bar{K} \quad (\text{二元时 } R = \bar{K}) \\
\text{香农码}&:\; k_i = \lceil -\log_2 p(a_i) \rceil,\; \text{码字} = \text{累加概率 }p_a\text{ 的二进制前 }k_i\text{ 位} \\
\text{费诺码}&:\; \text{按概率等分} \to \text{递归分配码元} \\
\text{赫夫曼码}&:\; \text{概率排序} \to \text{合并最小两个} \to \text{反向读码字} \\
\text{克拉夫特不等式}&:\; \sum m^{-k_i} \le 1 \quad \text{(即时码充要条件)}
\end{aligned}
$$

## 率失真函数

$$
\begin{aligned}
\bar{D} &= \sum\sum p(a_i)p(b_j \mid a_i) d(a_i, b_j) \\
R(D) &= \min I(X;Y) \quad \text{s.t. } \bar{D} \le D \quad \text{[注意是 min 不是 max!]} \\
R(D)&\text{ 下凸, 单调递减, 定义域 } [0, D_{\max}] \\
D_{\max} &= \min_j \sum p(a_i) d(a_i, b_j) \\
\text{二元汉明失真}(\alpha)&:\; R(D) = H(p) - H(D / \alpha),\; D_{\max} = \alpha \cdot \min(p, 1-p) \\
\text{等概率 }n\text{ 元汉明}&:\; R(D) = \log_2 n - H(D) - D \cdot \log_2(n-1),\; D_{\max} = (n-1) / n \\
\text{信息价值}&:\; V = D_{\max} - D(R)
\end{aligned}
$$

## 纠错编码

$$
\begin{aligned}
c &= m \cdot G \quad \text{(编码)},\quad s = r \cdot H^T \quad \text{(伴随式)} \\
G \cdot H^T &= 0 \quad (k \times (n-k)\text{ 零矩阵}) \\
\text{系统码}&:\; G_s = [I_k \mid Q],\; H_s = [Q^T \mid I_{n-k}] \\
d_{\min} &= \min w(c \neq 0) \quad \text{(非零码字)} \\
\text{检错}&:\; l = d_{\min} - 1,\quad \text{纠错}:\; t = \lfloor (d_{\min} - 1) / 2 \rfloor \\
\text{汉明码}&:\; n = 2^r - 1,\; k = 2^r - r - 1,\; d_{\min} = 3,\; t = 1 \\
\text{循环码}&:\; c(x) = m(x)g(x),\; g(x) \mid x^n - 1,\; \partial g(x) = n - k \\
h(x) &= (x^n - 1) / g(x) \quad \text{(校验多项式)} \\
\text{系统循环码}&:\; c(x) = x^{n-k} m(x) + \bigl[x^{n-k} m(x) \bmod g(x)\bigr] \\
\text{伴随多项式}&:\; s(x) = r(x) \bmod g(x) \\
x^7 - 1 &= (x + 1)(x^3 + x^2 + 1)(x^3 + x + 1) \quad [\mathrm{GF}(2)]
\end{aligned}
$$

## 密码

$$
\begin{aligned}
\text{完善保密性}&:\; I(P;C) = 0,\; \text{必要条件 } H(K) \ge H(P) \\
\text{唯一解距离}&:\; V \approx H(K) / I_{0\infty},\; I_{0\infty} = H_0 - H_1 \\
\text{加法密码}&:\; \text{有效密钥 }25,\; H(K) \approx 4.64,\; V \approx 6.36 \\
\text{乘法密码}&:\; \text{有效密钥 }11,\; H(K) \approx 3.46,\; V \approx 4.74 \quad (k\text{ 与 }26\text{ 互素}) \\
\text{仿射密码}&:\; \text{有效密钥 }275,\; H(K) \approx 8.10,\; V \approx 11.10
\end{aligned}
$$

## 马尔科夫信源

$$
\begin{aligned}
\text{平稳分布}&:\; p = p \cdot P,\; \sum p_i = 1 \quad \text{(解方程组)} \\
\text{极限熵}&:\; H_\infty = \sum p(s_i) \cdot H(X \mid s_i) = \sum p(s_i) \cdot \Bigl[-\sum p(x \mid s_i) \log_2 p(x \mid s_i)\Bigr]
\end{aligned}
$$

## C vs R(D)

$$
\begin{aligned}
C &= \max I(X;Y),\; \text{上凸},\; \text{对 } p(a_i),\; \text{仅与信道有关},\; \text{解决可靠性},\; \text{传输上界} \\
R(D) &= \min I(X;Y),\; \text{下凸},\; \text{对 } p(b_j \mid a_i),\; \text{仅与信源有关},\; \text{解决有效性},\; \text{压缩下界} \\
\text{成功通信条件}&:\; R(D) \le R \le C
\end{aligned}
$$

## 定理简述

$$
\begin{aligned}
\text{香农第一定理(无失真编码)}&:\; R > H(X) \to L\text{足够大可无失真} \\
\text{香农第二定理(有噪编码)}&:\; R < C \to L\text{足够大时 }P_e \to 0 \\
\text{香农第三定理(限失真编码)}&:\; R > R(D) \to L\text{足够大时 }\bar{D} \le D + \varepsilon \\
\text{最大离散熵定理}&:\; \text{等概时熵最大},\; H(X) \le \log_2 n
\end{aligned}
$$
